# model_chibi.txt 로 만든 2 등신 캐릭터에 입 + 표정 (셰이프 키) — nova model batch docs/examples/model_chibi_face.txt
# 셰이프 키 이름이 VRM 표정 이름 (happy · blink · aa …) 이면 export *.vrm 에서 그 표정이 된다 (엔진: Expressions 컴포넌트, 자동 깜빡임)
# 입: 납작한 구 (얼굴 앞)
add --type uvsphere --segments 16 --rings 8 --radius 0.03 --location 0,0.985,0.262 --name Mouth --smooth
object.transform --object Mouth --scale 1.6,0.45,0.5
mode --mode edit
select.all
material.set --index 5 --name Mouth --color 0.62,0.2,0.26
mode --mode object
# 눈 (EyeL = 캐릭터 왼쪽 = -X): blink · blinkLeft · happy (웃는 눈 = 납작하게 + 살짝 위)
object.select --name EyeL
mode --mode edit
select.all
shape.add --name blink
scale --factor 1,0.12,1
shape.add --name blinkLeft
scale --factor 1,0.12,1
shape.add --name happy
scale --factor 1.05,0.3,1
translate --delta 0,0.012,0
shape.edit --basis
mode --mode object
object.select --name EyeR
mode --mode edit
select.all
shape.add --name blink
scale --factor 1,0.12,1
shape.add --name blinkRight
scale --factor 1,0.12,1
shape.add --name happy
scale --factor 1.05,0.3,1
translate --delta 0,0.012,0
shape.edit --basis
mode --mode object
# 입 모양: aa ih ou ee oh + 웃음 (넓게)
object.select --name Mouth
mode --mode edit
select.all
shape.add --name aa
scale --factor 1.1,3.2,1
shape.add --name ih
scale --factor 1.5,1.3,1
shape.add --name ou
scale --factor 0.6,1.9,1
shape.add --name ee
scale --factor 1.35,1.7,1
shape.add --name oh
scale --factor 0.9,2.6,1
shape.add --name happy
scale --factor 1.6,1.5,1
shape.edit --basis
mode --mode object
